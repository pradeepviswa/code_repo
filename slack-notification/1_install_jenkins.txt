#!/bin/bash
set -e  # Exit immediately if a command fails

# Update package list
sudo apt update -y

# Install dependencies without asking for confirmation
sudo apt install -y fontconfig openjdk-21-jre wget gnupg

# Verify Java installation
java -version

# Add Jenkins key and repository
sudo mkdir -p /etc/apt/keyrings
sudo wget -q -O /etc/apt/keyrings/jenkins-keyring.asc \
  https://pkg.jenkins.io/debian-stable/jenkins.io-2023.key

echo "deb [signed-by=/etc/apt/keyrings/jenkins-keyring.asc] https://pkg.jenkins.io/debian-stable binary/" | \
  sudo tee /etc/apt/sources.list.d/jenkins.list > /dev/null

# Update package list again
sudo apt update -y

# Install Jenkins without prompts
sudo apt install -y jenkins

# Optionally, enable and start Jenkins service
sudo systemctl enable jenkins
sudo systemctl start jenkins

# Show Jenkins status
sudo systemctl status jenkins --no-pager

sudo apt install maven -y 

sudo cat /var/lib/jenkins/secrets/initialAdminPassword


#CONFIGURE BELOW IN JENKINS
# GLOBAL CONFGUREION -> TOOLS
#JDK -> JAVA_HOME → /usr/lib/jvm/java-21-openjdk-amd64
# MAVEN -> 
# Name: Maven-3.8.7
MAVEN_HOME → /usr/share/maven
